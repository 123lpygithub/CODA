def order_by_points(nums):

    def weight(x):
        x_list = list(str(x))
        if x_list[0] == '-':
            x_list = x_list[1:]
            x_list = list(map(int, x_list))
            x_list[0] = -x_list[0]
        else:
            x_list = list(map(int, x_list))
        return sum(x_list)
    return sorted(nums, key=weight)

def get_max_triples(n):
    if n <= 2:
        return False
    one_cnt = 1 + (n - 2) // 3 * 2 + (n - 2) % 3
    zero_cnt = n - one_cnt
    return one_cnt * (one_cnt - 1) * (one_cnt - 2) // 6 + zero_cnt * (zero_cnt - 1) * (zero_cnt - 2) // 6

def sorted_list_sum(lst):
    from functools import cmp_to_key

    def cmp(s: str, t: str):
        if len(s) != len(t):

        return -1 if s < t else 1

def Strongest_Extension(class_name, extensions):

    def strength(s: str) -> int:
        (CAP, SM) = (0, 0)
        for ch in s:
            if ch.isupper():
                CAP += 1
            if ch.islower():
                SM += 1
        return CAP - SM
    max_strength = max(map(strength, extensions))
    for e in extensions:
        if strength(e) == max_strength:
            return class_name + '.' + e

def Strongest_Extension(class_name, extensions):
    strength = []
    for i in extensions:
        cap = 0
        sm = 0
        for j in i:
            if j.isupper():
                cap += 1
            else:
                sm += 1
        strength.append(cap - sm)
    strength.sort()
    return class_name + '.' + extensions[strength.index(strength[0])]
